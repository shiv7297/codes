#include <mpi.h>
#include <stdio.h>

int main(int argc, char *argv[])
{
    int rank, size;
    int x = 10, sum;
    int arr[8] = {1,2,3,4,5,6,7,8};
    int recv[2];

    MPI_Status status;
    MPI_Request request;

    /* Initialize MPI */
    MPI_Init(&argc, &argv);

    /* Get rank and size */
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    /* ---------------- MPI_Send ---------------- */
    MPI_Send(&x, 1, MPI_INT, 1, 0, MPI_COMM_WORLD);

    /* ---------------- MPI_Recv ---------------- */
    MPI_Recv(&x, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &status);

    /* ---------------- MPI_Isend ---------------- */
    MPI_Isend(&x, 1, MPI_INT, 1, 0, MPI_COMM_WORLD, &request);

    /* ---------------- MPI_Irecv ---------------- */
    MPI_Irecv(&x, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &request);

    /* ---------------- MPI_Wait ---------------- */
    MPI_Wait(&request, &status);

    /* ---------------- MPI_Bcast ---------------- */
    MPI_Bcast(&x, 1, MPI_INT, 0, MPI_COMM_WORLD);

    /* ---------------- MPI_Scatter ---------------- */
    MPI_Scatter(arr, 2, MPI_INT,
                recv, 2, MPI_INT,
                0, MPI_COMM_WORLD);

    /* ---------------- MPI_Gather ---------------- */
    MPI_Gather(recv, 2, MPI_INT,
               arr, 2, MPI_INT,
               0, MPI_COMM_WORLD);

    /* ---------------- MPI_Reduce ---------------- */
    MPI_Reduce(&x, &sum,
               1, MPI_INT,
               MPI_SUM,
               0, MPI_COMM_WORLD);

    /* ---------------- MPI_Allreduce ---------------- */
    MPI_Allreduce(&x, &sum,
                  1, MPI_INT,
                  MPI_SUM,
                  MPI_COMM_WORLD);

    /* ---------------- MPI_Barrier ---------------- */
    MPI_Barrier(MPI_COMM_WORLD);

    /* ---------------- Finalize MPI ---------------- */
    MPI_Finalize();

    return 0;
}
